API
read-only endpoints
Public read-only API for the xnet ephemeral network. All endpoints return JSON (except /api/events, which is text/event-stream). ← back to status
GET /api/status
Full status: phase, cutover, migration, services, versions, endpoints, dashboards.
curl https://migrate.xmtp.run/api/status
Node.js example 
// Node.js 18+
const res = await fetch("https://migrate.xmtp.run/api/status");
if (!res.ok) throw new Error(`HTTP ${res.status}`);
const status = await res.json();
console.log("phase:", status.phase);
console.log("cutover:", status.cutover?.scheduled_utc);
console.log("migration tables:", Object.keys(status.migration?.message_types ?? {}));

GET /api/migration
Per-table v3 → d14n migration progress. Percentages and source/dest seq IDs.
curl https://migrate.xmtp.run/api/migration
Node.js example 
// Node.js 18+ — poll migration until complete
async function waitForMigration() {
  while (true) {
    const res = await fetch("https://migrate.xmtp.run/api/migration");
    const m = await res.json();
    const types = Object.entries(m.message_types ?? {});
    if (!types.length) { console.log("no progress yet"); }
    for (const [k, t] of types) {
      console.log(`  ${k.padEnd(20)} ${t.percent.toFixed(1)}%  (${t.dest_seq}/${t.source_seq})`);
    }
    const done = types.length > 0 && types.every(([, t]) => t.percent >= 100);
    if (done) return;
    await new Promise(r => setTimeout(r, 5000));
  }
}
await waitForMigration();

GET /api/events
Server-Sent Events stream. Pushes phase transitions (v3_running, cutover_started, migrating, migration_complete, d14n_active, teardown_imminent) as they happen. First frame is a connected event describing current state. Clients reconnect automatically on disconnect. 
curl -N https://migrate.xmtp.run/api/events
Node.js example 
// Node.js 18+ — streams SSE events, handles frames manually.
// For production use, prefer the `eventsource` npm package which handles
// reconnects, backoff, and Last-Event-ID automatically.
async function watchEvents() {
  const res = await fetch("https://migrate.xmtp.run/api/events", {
    headers: { accept: "text/event-stream" },
  });
  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";

  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });

    let idx;
    while ((idx = buffer.indexOf("\n\n")) !== -1) {
      const frame = buffer.slice(0, idx);
      buffer = buffer.slice(idx + 2);
      const dataLine = frame.split("\n").find(l => l.startsWith("data: "));
      if (!dataLine) continue;
      const event = JSON.parse(dataLine.slice(6));
      console.log(`[${event.type}] at ${event.timestamp_s}`);
      if (event.type === "v3_running") {
        // kick off tests...
      }
    }
  }
}
watchEvents().catch(err => {
  console.error("stream closed:", err);
  // reconnect with backoff in real code
});
